if(DEFINED ENV{CONDA_PREFIX})
    set(CONDA_PREFIX $ENV{CONDA_PREFIX})
    message(STATUS "Found conda environment: ${CONDA_PREFIX}")
    set(Python3_ROOT_DIR ${CONDA_PREFIX})
    set(Python3_EXECUTABLE ${CONDA_PREFIX}/bin/python)
endif()

# ============================================
# Python/pybind11
# ============================================
find_package(Python3 REQUIRED COMPONENTS Interpreter)

set(PYTHON_MAJOR ${Python3_VERSION_MAJOR})
set(PYTHON_MINOR ${Python3_VERSION_MINOR})
option(BUILD_PYTHON_MODULE "Build Python module with pybind11" ON)

if(BUILD_PYTHON_MODULE)
    find_package(Python3 COMPONENTS Interpreter Development REQUIRED)
    message(STATUS "Python executable found: ${Python3_EXECUTABLE}")

    set(pybind11_DIR ${CONDA_PREFIX}/lib/python${PYTHON_MAJOR}.${PYTHON_MINOR}/site-packages/pybind11/share/cmake/pybind11)
    find_package(pybind11 REQUIRED)

    execute_process(
        COMMAND ${Python3_EXECUTABLE} -c "import numpy; print(numpy.get_include())"
        OUTPUT_VARIABLE NUMPY_INCLUDE_DIR
        OUTPUT_STRIP_TRAILING_WHITESPACE
        ERROR_QUIET
    )
    
    if(NUMPY_INCLUDE_DIR)
        message(STATUS "NumPy include directory: ${NUMPY_INCLUDE_DIR}")
    endif()
endif()


set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(OpenMP REQUIRED)
if(OpenMP_CXX_FOUND)
    message(STATUS "OpenMP found: ${OpenMP_CXX_FLAGS}")
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${OpenMP_CXX_FLAGS}")
endif()

set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/dual_contouring_c)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/dual_contouring_c/lib)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/dual_contouring_c/bin)

file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/dual_contouring_c)
file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/dual_contouring_c/lib)
file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/dual_contouring_c/bin)


if(BUILD_PYTHON_MODULE)

    add_library(py_graph_grouping MODULE 
    src/py_api.cpp
    src/sqef.cpp
    src/tangent.cpp     
    src/dbscan.cpp 
    src/plane_grouping.cpp
    src/SegmentOcclusionCheck.cpp
    src/cloud_rvd.cpp
    src/octree_node.cpp
    src/octree.cpp
    )

    set_target_properties(py_graph_grouping PROPERTIES
        OUTPUT_NAME "GRAPH_GROUPING"
        PREFIX ""
        SUFFIX ".so"
        CXX_STANDARD 17
        CXX_STANDARD_REQUIRED ON
        CXX_EXTENSIONS OFF
    )

    target_include_directories(py_graph_grouping PRIVATE
        ${Python3_INCLUDE_DIRS}
        ${pybind11_INCLUDE_DIR}
        ${NUMPY_INCLUDE_DIR}
        ${CMAKE_CURRENT_SOURCE_DIR}/include
    )
    
    target_link_libraries(py_graph_grouping PRIVATE
        pybind11::module
        ${OpenMP_CXX_LIBRARIES}
    )

    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(py_graph_grouping PRIVATE
            -O3
            -Wall
            -Wextra
            -fvisibility=hidden
            -fPIC
            ${OpenMP_CXX_FLAGS}
        )
    endif()
endif()

install(DIRECTORY include/ DESTINATION include)
